   //------- authSendEmail - PHP 5 / SMTP SSL Port 465 -------

    public function authSendEmail($from, $namefrom, $to, $nameto, $cc, $subject, $message)
    {
        /*
     * ============================================================
     * SMTP CONFIGURATION
     * ============================================================
     */

        $smtpServer = "webmail.fiberathome.net";
        $port       = 465;

        // New SMTP credentials
        $username   = "itm1@fiberathome.net";
        $password   = '80yi!SDLW@$cXp';

        $timeout    = 60;
        $localhost  = "localhost";
        $newLine    = "\r\n";


        /*
     * ============================================================
     * DATABASE CONFIGURATION
     * ============================================================
     */

        $dbHost = "localhost";
        $dbUser = "root1";
        $dbPass = "F@hitmrt1";
        $dbName = "itm";


        /*
     * ============================================================
     * USER INFORMATION
     * ============================================================
     */

        $user_ip = '';

        if (isset($_SERVER['REMOTE_ADDR'])) {
            $user_ip = $_SERVER['REMOTE_ADDR'];
        }


        $session_user = '';

        if (isset($_SESSION['eID'])) {
            $session_user = $_SESSION['eID'];
        }


        /*
     * ============================================================
     * SANITIZE HEADER VALUES
     * ============================================================
     */

        $namefrom = str_replace(
            array("\r", "\n"),
            '',
            $namefrom
        );

        $nameto = str_replace(
            array("\r", "\n"),
            '',
            $nameto
        );

        $subject = str_replace(
            array("\r", "\n"),
            '',
            $subject
        );

        $from = str_replace(
            array("\r", "\n"),
            '',
            $from
        );

        $to = str_replace(
            array("\r", "\n"),
            '',
            $to
        );

        $cc = str_replace(
            array("\r", "\n"),
            '',
            $cc
        );


        /*
     * ============================================================
     * SMTP STATUS
     * ============================================================
     */

        $mailStatus = "fail";
        $errorMessage = "";
        $smtpConnect = false;


        /*
     * ============================================================
     * CONNECT USING SSL
     *
     * Port 465 uses implicit SSL.
     * DO NOT send STARTTLS.
     * ============================================================
     */

        $smtpConnect = @fsockopen(
            "ssl://" . $smtpServer,
            $port,
            $errno,
            $errstr,
            $timeout
        );


        if (!$smtpConnect) {

            $errorMessage =
                "SMTP connection failed: " .
                $errno .
                " - " .
                $errstr;
        } else {

            /*
         * Socket timeout
         */
            stream_set_timeout(
                $smtpConnect,
                $timeout
            );


            /*
         * ========================================================
         * READ SMTP WELCOME MESSAGE
         * ========================================================
         */

            $smtpResponse = "";

            while (!feof($smtpConnect)) {

                $line = fgets(
                    $smtpConnect,
                    515
                );

                if ($line === false) {
                    break;
                }

                $smtpResponse .= $line;

                if (
                    strlen($line) >= 4 &&
                    substr($line, 3, 1) == " "
                ) {
                    break;
                }
            }


            $smtpCode = intval(
                substr(
                    $smtpResponse,
                    0,
                    3
                )
            );


            if ($smtpCode != 220) {

                $errorMessage =
                    "SMTP server connection rejected: " .
                    $smtpResponse;
            } else {

                /*
             * ====================================================
             * EHLO
             * ====================================================
             */

                fputs(
                    $smtpConnect,
                    "EHLO " .
                        $localhost .
                        $newLine
                );


                $smtpResponse = "";

                while (!feof($smtpConnect)) {

                    $line = fgets(
                        $smtpConnect,
                        515
                    );

                    if ($line === false) {
                        break;
                    }

                    $smtpResponse .= $line;

                    if (
                        strlen($line) >= 4 &&
                        substr($line, 3, 1) == " "
                    ) {
                        break;
                    }
                }


                $smtpCode = intval(
                    substr(
                        $smtpResponse,
                        0,
                        3
                    )
                );


                /*
             * If EHLO fails, try HELO.
             */
                if ($smtpCode != 250) {

                    fputs(
                        $smtpConnect,
                        "HELO " .
                            $localhost .
                            $newLine
                    );


                    $smtpResponse = "";

                    while (!feof($smtpConnect)) {

                        $line = fgets(
                            $smtpConnect,
                            515
                        );

                        if ($line === false) {
                            break;
                        }

                        $smtpResponse .= $line;

                        if (
                            strlen($line) >= 4 &&
                            substr($line, 3, 1) == " "
                        ) {
                            break;
                        }
                    }


                    $smtpCode = intval(
                        substr(
                            $smtpResponse,
                            0,
                            3
                        )
                    );
                }


                if ($smtpCode != 250) {

                    $errorMessage =
                        "SMTP EHLO/HELO failed: " .
                        $smtpResponse;
                } else {

                    /*
                 * =================================================
                 * AUTH LOGIN
                 * =================================================
                 */

                    fputs(
                        $smtpConnect,
                        "AUTH LOGIN" .
                            $newLine
                    );


                    $smtpResponse = fgets(
                        $smtpConnect,
                        515
                    );


                    $smtpCode = intval(
                        substr(
                            $smtpResponse,
                            0,
                            3
                        )
                    );


                    if ($smtpCode != 334) {

                        $errorMessage =
                            "SMTP AUTH LOGIN failed: " .
                            $smtpResponse;
                    } else {

                        /*
                     * =============================================
                     * SMTP USERNAME
                     * =============================================
                     */

                        fputs(
                            $smtpConnect,
                            base64_encode(
                                $username
                            ) .
                                $newLine
                        );


                        $smtpResponse = fgets(
                            $smtpConnect,
                            515
                        );


                        $smtpCode = intval(
                            substr(
                                $smtpResponse,
                                0,
                                3
                            )
                        );


                        if ($smtpCode != 334) {

                            $errorMessage =
                                "SMTP username rejected: " .
                                $smtpResponse;
                        } else {

                            /*
                         * =========================================
                         * SMTP PASSWORD
                         * =========================================
                         */

                            fputs(
                                $smtpConnect,
                                base64_encode(
                                    $password
                                ) .
                                    $newLine
                            );


                            $smtpResponse = fgets(
                                $smtpConnect,
                                515
                            );


                            $smtpCode = intval(
                                substr(
                                    $smtpResponse,
                                    0,
                                    3
                                )
                            );


                            if ($smtpCode != 235) {

                                $errorMessage =
                                    "SMTP authentication failed: " .
                                    $smtpResponse;
                            } else {

                                /*
                             * =====================================
                             * MAIL FROM
                             *
                             * Use authenticated SMTP account.
                             * =====================================
                             */

                                fputs(
                                    $smtpConnect,
                                    "MAIL FROM:<" .
                                        $username .
                                        ">" .
                                        $newLine
                                );


                                $smtpResponse = fgets(
                                    $smtpConnect,
                                    515
                                );


                                $smtpCode = intval(
                                    substr(
                                        $smtpResponse,
                                        0,
                                        3
                                    )
                                );


                                if (
                                    $smtpCode != 250 &&
                                    $smtpCode != 251
                                ) {

                                    $errorMessage =
                                        "MAIL FROM rejected: " .
                                        $smtpResponse;
                                } else {

                                    /*
                                 * =================================
                                 * TO RECIPIENTS
                                 *
                                 * Supports:
                                 *
                                 * abc@test.com
                                 *
                                 * abc@test.com,xyz@test.com
                                 *
                                 * abc@test.com;xyz@test.com
                                 * =================================
                                 */

                                    $toRecipients = preg_split(
                                        '/[;,]+/',
                                        $to
                                    );


                                    $recipientError = false;


                                    foreach (
                                        $toRecipients
                                        as $recipient
                                    ) {

                                        $recipient = trim(
                                            $recipient
                                        );


                                        if ($recipient == '') {
                                            continue;
                                        }


                                        fputs(
                                            $smtpConnect,
                                            "RCPT TO:<" .
                                                $recipient .
                                                ">" .
                                                $newLine
                                        );


                                        $smtpResponse = fgets(
                                            $smtpConnect,
                                            515
                                        );


                                        $smtpCode = intval(
                                            substr(
                                                $smtpResponse,
                                                0,
                                                3
                                            )
                                        );


                                        if (
                                            $smtpCode != 250 &&
                                            $smtpCode != 251
                                        ) {

                                            $errorMessage =
                                                "TO recipient rejected (" .
                                                $recipient .
                                                "): " .
                                                $smtpResponse;

                                            $recipientError = true;

                                            break;
                                        }
                                    }


                                    /*
                                 * =================================
                                 * CC RECIPIENTS
                                 * =================================
                                 */

                                    $ccRecipients = array();


                                    if (
                                        !$recipientError &&
                                        trim($cc) != ''
                                    ) {

                                        $ccRecipients = preg_split(
                                            '/[;,]+/',
                                            $cc
                                        );


                                        foreach (
                                            $ccRecipients
                                            as $recipient
                                        ) {

                                            $recipient = trim(
                                                $recipient
                                            );


                                            if ($recipient == '') {
                                                continue;
                                            }


                                            fputs(
                                                $smtpConnect,
                                                "RCPT TO:<" .
                                                    $recipient .
                                                    ">" .
                                                    $newLine
                                            );


                                            $smtpResponse = fgets(
                                                $smtpConnect,
                                                515
                                            );


                                            $smtpCode = intval(
                                                substr(
                                                    $smtpResponse,
                                                    0,
                                                    3
                                                )
                                            );


                                            if (
                                                $smtpCode != 250 &&
                                                $smtpCode != 251
                                            ) {

                                                $errorMessage =
                                                    "CC recipient rejected (" .
                                                    $recipient .
                                                    "): " .
                                                    $smtpResponse;

                                                $recipientError = true;

                                                break;
                                            }
                                        }
                                    }


                                    if (!$recipientError) {

                                        /*
                                     * =============================
                                     * DATA
                                     * =============================
                                     */

                                        fputs(
                                            $smtpConnect,
                                            "DATA" .
                                                $newLine
                                        );


                                        $smtpResponse = fgets(
                                            $smtpConnect,
                                            515
                                        );


                                        $smtpCode = intval(
                                            substr(
                                                $smtpResponse,
                                                0,
                                                3
                                            )
                                        );


                                        if ($smtpCode != 354) {

                                            $errorMessage =
                                                "SMTP DATA rejected: " .
                                                $smtpResponse;
                                        } else {

                                            /*
                                         * =========================
                                         * EMAIL HEADERS
                                         * =========================
                                         */

                                            $encodedSubject =
                                                "=?UTF-8?B?" .
                                                base64_encode(
                                                    $subject
                                                ) .
                                                "?=";


                                            $encodedFromName =
                                                "=?UTF-8?B?" .
                                                base64_encode(
                                                    $namefrom
                                                ) .
                                                "?=";


                                            $encodedToName =
                                                "=?UTF-8?B?" .
                                                base64_encode(
                                                    $nameto
                                                ) .
                                                "?=";


                                            /*
                                         * Generate Message ID
                                         */
                                            $domain = $smtpServer;

                                            if (
                                                strpos(
                                                    $username,
                                                    "@"
                                                ) !== false
                                            ) {

                                                $tempDomain =
                                                    explode(
                                                        "@",
                                                        $username
                                                    );

                                                if (
                                                    isset(
                                                        $tempDomain[1]
                                                    )
                                                ) {

                                                    $domain =
                                                        $tempDomain[1];
                                                }
                                            }


                                            $messageID =
                                                "<" .
                                                md5(
                                                    uniqid(
                                                        mt_rand(),
                                                        true
                                                    )
                                                ) .
                                                "@" .
                                                $domain .
                                                ">";


                                            /*
                                         * =========================
                                         * BUILD HEADER
                                         * =========================
                                         */

                                            $headers = "";

                                            $headers .=
                                                "Date: " .
                                                date("r") .
                                                $newLine;

                                            $headers .=
                                                "Message-ID: " .
                                                $messageID .
                                                $newLine;

                                            $headers .=
                                                "MIME-Version: 1.0" .
                                                $newLine;

                                            $headers .=
                                                "Content-Type: text/html; charset=UTF-8" .
                                                $newLine;

                                            $headers .=
                                                "Content-Transfer-Encoding: 8bit" .
                                                $newLine;


                                            /*
                                         * SMTP authenticated account
                                         * is used as actual From.
                                         */
                                            $headers .=
                                                "From: " .
                                                $encodedFromName .
                                                " <" .
                                                $username .
                                                ">" .
                                                $newLine;


                                            /*
                                         * If original $from is
                                         * different, use Reply-To.
                                         */
                                            if (
                                                trim($from) != '' &&
                                                strtolower($from) != strtolower($username)
                                            ) {

                                                $headers .=
                                                    "Reply-To: " .
                                                    $from .
                                                    $newLine;
                                            }


                                            /*
                                         * TO header
                                         */
                                            if (
                                                count($toRecipients) == 1 &&
                                                trim($nameto) != ''
                                            ) {

                                                $headers .=
                                                    "To: " .
                                                    $encodedToName .
                                                    " <" .
                                                    trim(
                                                        $toRecipients[0]
                                                    ) .
                                                    ">" .
                                                    $newLine;
                                            } else {

                                                $cleanTo = array();

                                                foreach (
                                                    $toRecipients
                                                    as $emailAddress
                                                ) {

                                                    $emailAddress =
                                                        trim(
                                                            $emailAddress
                                                        );

                                                    if (
                                                        $emailAddress != ''
                                                    ) {

                                                        $cleanTo[] =
                                                            $emailAddress;
                                                    }
                                                }


                                                $headers .=
                                                    "To: " .
                                                    implode(
                                                        ", ",
                                                        $cleanTo
                                                    ) .
                                                    $newLine;
                                            }


                                            /*
                                         * CC header
                                         */
                                            if (
                                                count(
                                                    $ccRecipients
                                                ) > 0
                                            ) {

                                                $cleanCC = array();

                                                foreach (
                                                    $ccRecipients
                                                    as $emailAddress
                                                ) {

                                                    $emailAddress =
                                                        trim(
                                                            $emailAddress
                                                        );

                                                    if (
                                                        $emailAddress != ''
                                                    ) {

                                                        $cleanCC[] =
                                                            $emailAddress;
                                                    }
                                                }


                                                if (
                                                    count(
                                                        $cleanCC
                                                    ) > 0
                                                ) {

                                                    $headers .=
                                                        "Cc: " .
                                                        implode(
                                                            ", ",
                                                            $cleanCC
                                                        ) .
                                                        $newLine;
                                                }
                                            }


                                            $headers .=
                                                "Subject: " .
                                                $encodedSubject .
                                                $newLine;


                                            /*
                                         * =========================
                                         * NORMALIZE BODY LINE ENDINGS
                                         * =========================
                                         */

                                            $smtpMessage =
                                                str_replace(
                                                    array(
                                                        "\r\n",
                                                        "\r"
                                                    ),
                                                    "\n",
                                                    $message
                                                );


                                            $smtpMessage =
                                                str_replace(
                                                    "\n",
                                                    $newLine,
                                                    $smtpMessage
                                                );


                                            /*
                                         * SMTP dot-stuffing.
                                         *
                                         * Any body line beginning
                                         * with "." needs another ".".
                                         */
                                            $smtpMessage =
                                                preg_replace(
                                                    '/^\./m',
                                                    '..',
                                                    $smtpMessage
                                                );


                                            /*
                                         * =========================
                                         * SEND EMAIL
                                         * =========================
                                         */

                                            $emailData =
                                                $headers .
                                                $newLine .
                                                $smtpMessage .
                                                $newLine .
                                                "." .
                                                $newLine;


                                            fputs(
                                                $smtpConnect,
                                                $emailData
                                            );


                                            /*
                                         * =========================
                                         * SMTP FINAL RESPONSE
                                         * =========================
                                         */

                                            $smtpResponse = "";

                                            while (
                                                !feof(
                                                    $smtpConnect
                                                )
                                            ) {

                                                $line = fgets(
                                                    $smtpConnect,
                                                    515
                                                );

                                                if (
                                                    $line === false
                                                ) {
                                                    break;
                                                }

                                                $smtpResponse .=
                                                    $line;

                                                if (
                                                    strlen(
                                                        $line
                                                    ) >= 4 &&
                                                    substr(
                                                        $line,
                                                        3,
                                                        1
                                                    ) == " "
                                                ) {
                                                    break;
                                                }
                                            }


                                            $smtpCode = intval(
                                                substr(
                                                    $smtpResponse,
                                                    0,
                                                    3
                                                )
                                            );


                                            if (
                                                $smtpCode == 250
                                            ) {

                                                $mailStatus =
                                                    "success";
                                            } else {

                                                $errorMessage =
                                                    "Email rejected by SMTP server: " .
                                                    $smtpResponse;
                                            }
                                        }
                                    }
                                }
                            }
                        }
                    }
                }
            }


            /*
         * ========================================================
         * CLOSE SMTP CONNECTION
         * ========================================================
         */

            if (
                is_resource(
                    $smtpConnect
                )
            ) {

                @fputs(
                    $smtpConnect,
                    "QUIT" .
                        $newLine
                );

                @fclose(
                    $smtpConnect
                );
            }
        }


        /*
     * ============================================================
     * DATABASE EMAIL LOG
     * ============================================================
     */

        $conn = @mysql_connect(
            $dbHost,
            $dbUser,
            $dbPass
        );


        if ($conn) {

            @mysql_select_db(
                $dbName,
                $conn
            );


            /*
         * Escape DB values
         */
            $dbFrom =
                mysql_real_escape_string(
                    $username,
                    $conn
                );

            $dbFromName =
                mysql_real_escape_string(
                    $namefrom,
                    $conn
                );

            $dbTo =
                mysql_real_escape_string(
                    $to,
                    $conn
                );

            $dbToName =
                mysql_real_escape_string(
                    $nameto,
                    $conn
                );

            $dbCc =
                mysql_real_escape_string(
                    $cc,
                    $conn
                );

            $dbSubject =
                mysql_real_escape_string(
                    $subject,
                    $conn
                );


            /*
         * Preserve the same behavior as
         * your old application.
         */
            $messageData =
                mysql_real_escape_string(
                    htmlspecialchars(
                        $message,
                        ENT_QUOTES,
                        "UTF-8"
                    ),
                    $conn
                );


            $dbSessionUser =
                mysql_real_escape_string(
                    $session_user,
                    $conn
                );

            $dbUserIp =
                mysql_real_escape_string(
                    $user_ip,
                    $conn
                );


            /*
         * Insert exactly one mail log.
         */
            $sql = "
            INSERT INTO tbl_emails
            (
                `from`,
                `from_name`,
                `to`,
                `to_name`,
                `cc`,
                `subject`,
                `body`,
                `mail_status`,
                `session_user`,
                `session_ip`
            )
            VALUES
            (
                '" . $dbFrom . "',
                '" . $dbFromName . "',
                '" . $dbTo . "',
                '" . $dbToName . "',
                '" . $dbCc . "',
                '" . $dbSubject . "',
                '" . $messageData . "',
                '" . $mailStatus . "',
                '" . $dbSessionUser . "',
                '" . $dbUserIp . "'
            )
        ";


            @mysql_query(
                $sql,
                $conn
            );
        }


        /*
     * ============================================================
     * RETURN
     * ============================================================
     */

        if ($mailStatus == "success") {

            return true;
        } else {

            return $errorMessage;
        }
    } // end authSendEmail
